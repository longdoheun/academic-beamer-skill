\NewDocumentEnvironment{GridRow}{O{}+b}{
  \bool_if:NT \l_ab_inrow_bool
    {\msg_error:nnn {academic-beamer}{invalid}{nested~GridRows~are~unsupported}}
  \group_begin:
  \bool_set_true:N \l_ab_inrow_bool
  \keys_set:nn {academic-beamer/row}{#1}
  \dim_set:Nn \l_ab_usable_dim {\linewidth}
  \ab_calculate:
  \int_gzero:N \g_ab_used_int
  \hbox_set:Nn \l_ab_row_box {
    \ignorespaces #2 \unskip
    \int_compare:nNnF {\g_ab_used_int} = {\l_ab_columns_int}
      {\msg_error:nnn {academic-beamer}{invalid}{row~spans~must~sum~to~columns}}
  }
  \dim_compare:nNnT {\box_wd:N \l_ab_row_box} > {\l_ab_usable_dim+0.01pt}
    {\msg_error:nnn {academic-beamer}{invalid}{row~width~exceeds~usable~width}}
  \dim_compare:nNnT {\box_ht:N \l_ab_row_box+\box_dp:N \l_ab_row_box} > {\l_ab_maxheight_dim}
    {\msg_error:nnn {academic-beamer}{invalid}{row~height~exceeds~maximum}}
  \iow_term:x {ABGRID:~n=\int_use:N \l_ab_columns_int;
    ~usable=\dim_use:N \l_ab_usable_dim;~track=\dim_use:N \l_ab_track_dim;
    ~actual=\dim_eval:n {\box_wd:N \l_ab_row_box}}
  \par\noindent\box_use:N \l_ab_row_box\par\vspace{\l_ab_rowgap_dim}
  \group_end:
}{ }

    \dim_compare_p:nNn {\l_ab_nativew_dim} = {0pt} &&
    \dim_compare_p:nNn {\l_ab_nativeh_dim} = {0pt} &&
    \fp_compare_p:nNn {\l_ab_nativefont_fp} = {0}
  }
    {\iow_term:n {ABFIGURE:~font~readability~unverified;~inspect~render}}
    {
      \fp_compare:nNnT {min(\dim_to_fp:n {\l_ab_nativew_dim},\dim_to_fp:n {\l_ab_nativeh_dim},\l_ab_nativefont_fp,\l_ab_minfont_fp)} < {0.001}

        {\msg_error:nnn {academic-beamer}{invalid}{figure~labels~would~be~unreadable}}
    }

  \group_end:
}
