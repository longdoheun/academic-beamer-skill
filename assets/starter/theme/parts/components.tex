\newcommand{\ABBlockBegin}[2]{%
  \par\vspace{1.4mm}\setlength{\ABBlockWidth}{\linewidth}%
  \usebeamercolor{#1}\colorlet{ABTitleBackground}{bg}\colorlet{ABTitleForeground}{fg}%
  \usebeamercolor{#2}\colorlet{ABBodyBackground}{bg}\colorlet{ABBodyForeground}{fg}%
  \ifdim\dimexpr\ABBlockWidth-2\ABPaddingX\relax>0pt\else
    \PackageError{academic-beamer}{Block padding exceeds width}{Give the block a wider span.}\fi
  \begin{lrbox}{\ABBlockTitleBox}\begin{minipage}{\dimexpr\ABBlockWidth-2\ABPaddingX\relax}
    \usebeamerfont{#1}\color{ABTitleForeground}\raggedright\insertblocktitle\par
  \end{minipage}\end{lrbox}%
  \begin{lrbox}{\ABBlockBodyBox}\begin{minipage}{\dimexpr\ABBlockWidth-2\ABPaddingX\relax}
    \usebeamerfont{#2}\ABInBlocktrue\ABBlockKoreanFont
    \color{ABBodyForeground}\setlength{\leftmargini}{1.1em}%
}
\newcommand{\ABBlockEnd}{%
  \par\end{minipage}\end{lrbox}%
  \setlength{\ABTitleHeight}{\dimexpr\ht\ABBlockTitleBox+\dp\ABBlockTitleBox+2\ABPaddingTitle\relax}%
  \setlength{\ABBodyHeight}{\dimexpr\ht\ABBlockBodyBox+\dp\ABBlockBodyBox+2\ABPaddingBody\relax}%
  \setlength{\ABTotalHeight}{\dimexpr\ABTitleHeight+\ABBodyHeight\relax}%
  \noindent\begin{tikzpicture}
    \begin{scope}\clip[rounded corners=\ABRadius](0,0)rectangle(\ABBlockWidth,-\ABTotalHeight);
      \fill[ABBodyBackground](0,0)rectangle(\ABBlockWidth,-\ABTotalHeight);
      \fill[ABTitleBackground](0,0)rectangle(\ABBlockWidth,-\ABTitleHeight);\end{scope}
    \draw[draw=ABOutline,line width=.4pt,rounded corners=\ABRadius]
      (.2pt,-.2pt)rectangle(\ABBlockWidth-.2pt,-\ABTotalHeight+.2pt);
    \node[anchor=north west,inner sep=0pt,outer sep=0pt]at(\ABPaddingX,-\ABPaddingTitle){\usebox{\ABBlockTitleBox}};
    \node[anchor=north west,inner sep=0pt,outer sep=0pt]at(\ABPaddingX,-\ABTitleHeight-\ABPaddingBody){\usebox{\ABBlockBodyBox}};
  \end{tikzpicture}\par\vspace{1.2mm}}
